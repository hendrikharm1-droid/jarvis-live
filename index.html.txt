  <script>
    // =========================================================================
    // APPS SCRIPT ENDPOINT (Kein API-Key im Quellcode nötig)
    // =========================================================================
    const APPS_SCRIPT_WEBAPP_URL = "https://script.google.com/macros/s/AKfycbyS3rW6ffr5STE7ih5NB3F1bpzmgbwgMAI4vlXFk0KR910EfV-F7OoNXP23H4XFMJXu9w/exec";
    // =========================================================================

    let isConnected = false;
    let ws = null;
    let audioContext = null;
    let mediaStream = null;
    let authToken = localStorage.getItem("JARVIS_AUTH_TOKEN");
    let dynamicApiKey = null;

    const orb = document.getElementById("voiceOrb");
    const statusText = document.getElementById("statusText");

    function log(msg) {
      const b = document.getElementById("logBox");
      b.innerHTML += `<br>• ${msg}`;
      b.scrollTop = b.scrollHeight;
    }

    window.addEventListener("DOMContentLoaded", () => {
      if (!authToken) {
        document.getElementById("authOverlay").style.display = "flex";
      }
    });

    function saveAuthToken() {
      const val = document.getElementById("tokenInput").value.trim();
      if (val) {
        localStorage.setItem("JARVIS_AUTH_TOKEN", val);
        authToken = val;
        document.getElementById("authOverlay").style.display = "none";
        log("✅ Schlüssel gespeichert.");
      }
    }

    orb.addEventListener("click", handleOrbClick);

    async function handleOrbClick() {
      if (!authToken) {
        document.getElementById("authOverlay").style.display = "flex";
        return;
      }
      if (isConnected) {
        stopSession();
      } else {
        await startSession();
      }
    }

    async function startSession() {
      orb.classList.add("active");
      statusText.innerText = "Initialisiere...";
      log("Frage Mikrofon-Hardware an...");

      try {
        mediaStream = await navigator.mediaDevices.getUserMedia({
          audio: { sampleRate: 16000, channelCount: 1, echoCancellation: true, noiseSuppression: true }
        });

        audioContext = new (window.AudioContext || window.webkitAudioContext)({ sampleRate: 16000 });
        if (audioContext.state === 'suspended') {
          await audioContext.resume();
        }

        log("🎙️ Mikrofon freigegeben.");
        statusText.innerText = "Autorisiere Sitzung...";

        // API Key dynamisch und sicher vom Backend abrufen
        if (!dynamicApiKey) {
          log("Hole Session-Token vom Backend...");
          const res = await fetch(APPS_SCRIPT_WEBAPP_URL, {
            method: "POST",
            body: JSON.stringify({ authToken: authToken, action: "get_config" })
          }).then(r => r.json());

          if (res.error) {
            log("🚨 Token ungültig!");
            localStorage.removeItem("JARVIS_AUTH_TOKEN");
            authToken = null;
            document.getElementById("authOverlay").style.display = "flex";
            stopSession();
            return;
          }
          dynamicApiKey = res.apiKey;
        }

        connectWebSocket();
      } catch (err) {
        log("❌ Fehler: " + err.name + " - " + err.message);
        statusText.innerText = "Fehler aufgetreten.";
        stopSession();
      }
    }

    function connectWebSocket() {
      log("Verbinde mit Gemini Live...");
      const tools = [{
        functionDeclarations: [
          {
            name: "search_documents",
            description: "Durchsucht Drive Index & Vektoren nach Dokumenten/Rechnungen.",
            parameters: { type: "OBJECT", properties: { query: { type: "STRING" } }, required: ["query"] }
          },
          {
            name: "get_upcoming_deadlines",
            description: "Holt Zahlungsziele und Fristen aus dem Google Kalender.",
            parameters: { type: "OBJECT", properties: { days: { type: "NUMBER" } } }
          },
          {
            name: "get_tax_summary",
            description: "Liest Steuerwerte (§35a Handwerkerlohn) aus.",
            parameters: { type: "OBJECT", properties: { category: { type: "STRING" } } }
          }
        ]
      }];

      const wsUrl = `wss://generativelanguage.googleapis.com/ws/google.ai.generativelanguage.v1alpha.GenerativeService.BidiGenerateContent?key=${dynamicApiKey}`;
      ws = new WebSocket(wsUrl);

      ws.onopen = () => {
        log("✅ WebSocket gesichert aktiv.");
        statusText.innerText = "JARVIS hört zu...";
        ws.send(JSON.stringify({
          setup: {
            model: "models/gemini-2.0-flash-exp",
            generationConfig: { responseModalities: ["AUDIO"] },
            tools: tools,
            systemInstruction: { parts: [{ text: "Du bist JARVIS, der persönliche Assistent von Hendrik Harm. Antworte kurz, prägnant auf Deutsch. Nutze deine Tools proaktiv." }] }
          }
        }));
        isConnected = true;
        startAudioStreaming();
      };

      ws.onmessage = async (event) => {
        const res = JSON.parse(event.data);
        if (res.serverContent?.modelTurn?.parts) {
          for (let p of res.serverContent.modelTurn.parts) {
            if (p.functionCall) {
              const call = p.functionCall;
              log(`⚡ Tool Call: <b>${call.name}</b>`);
              
              try {
                const apiRes = await fetch(APPS_SCRIPT_WEBAPP_URL, {
                  method: "POST",
                  body: JSON.stringify({ 
                    authToken: authToken, 
                    action: call.name, 
                    params: call.args || {} 
                  })
                }).then(r => r.json());

                ws.send(JSON.stringify({
                  toolResponse: { functionResponses: [{ response: { output: apiRes }, id: call.id }] }
                }));
              } catch(e) {
                log("⚠️ Tool-Fehler: " + e.message);
              }
            }
          }
        }
      };

      ws.onerror = (e) => log("❌ WebSocket Fehler");
      ws.onclose = () => stopSession();
    }

    function startAudioStreaming() {
      const source = audioContext.createMediaStreamSource(mediaStream);
      const processor = audioContext.createScriptProcessor(4096, 1, 1);

      processor.onaudioprocess = (e) => {
        if (!isConnected || ws.readyState !== WebSocket.OPEN) return;
        const input = e.inputBuffer.getChannelData(0);
        const pcm16 = new Int16Array(input.length);
        for (let i = 0; i < input.length; i++) {
          pcm16[i] = Math.max(-1, Math.min(1, input[i])) * 0x7FFF;
        }
        const base64Audio = btoa(String.fromCharCode(...new Uint8Array(pcm16.buffer)));
        ws.send(JSON.stringify({
          realtimeInput: { mediaChunks: [{ mimeType: "audio/pcm;rate=16000", data: base64Audio }] }
        }));
      };

      source.connect(processor);
      processor.connect(audioContext.destination);
      log("🎙️ Live Stream aktiv.");
    }

    function stopSession() {
      isConnected = false;
      if (ws) { try { ws.close(); } catch(e){} }
      if (mediaStream) { mediaStream.getTracks().forEach(t => t.stop()); }
      if (audioContext && audioContext.state !== 'closed') { try { audioContext.close(); } catch(e){} }
      orb.classList.remove("active");
      statusText.innerText = "Tippe auf den Orb, um zu sprechen";
      log("Sitzung beendet.");
    }
  </script>
